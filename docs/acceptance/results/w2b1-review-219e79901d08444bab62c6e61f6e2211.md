真实候选外层代码（命令转录，不在文档动作中执行）：

```python
import io, json, uuid
from contextlib import redirect_stdout, redirect_stderr
from datainsight import planning as p, feasibility as f
env = f.environment()
token = uuid.uuid4().hex
observation = p.new_run(p.OUTPUT / ("w2b1-local-observation-" + token))
target = p.OUTPUT / ("w2b1-candidates-" + token)
context = p.load_s2_accept(p.S2)
files = p.planning_manifest(context)["files"]
def identities():
    return [{"path": v["path"], "sha256": f.digest(f.read_bytes(f.ROOT / v["path"] ))} for v in files]
before = identities()
p.save(observation / "protected-before.json", before)
p.save(observation / "environment.json", env)
argv = ["candidates", "--output-dir", str(target)]
out, err = io.StringIO(), io.StringIO()
with redirect_stdout(out), redirect_stderr(err):
    status = p.main(argv)
after = identities()
p.save(observation / "protected-after.json", after)
findings = []
artifacts = []
if target.exists():
    for path in sorted(target.iterdir()):
        f.path_check(path, p.OUTPUT)
        if not path.is_file() or path.suffix not in (".json", ".tmp"):
            raise RuntimeError("unexpected_candidate_artifact")
        data = path.read_bytes()
        artifacts.append({"path": f.relative(path), "size": len(data), "sha256": f.digest(data)})
        for marker in ("SOURCE", "ROW", "IDENTIFIER", "SECRET", "INSTRUCTION"):
            if ("W2A_R1R4_" + marker + "_SENTINEL").encode() in data:
                findings.append({"path": f.relative(path), "marker_id": marker})
p.save(observation / "operation.json", {"main_argv": argv, "stdout": out.getvalue(), "stderr": err.getvalue(), "cli_exit_status": status, "protection_unchanged": before == after, "artifacts": artifacts, "sentinel_findings": findings, "scan_scope": "fixed synthetic markers in candidate files only; not a general privacy audit", "final_acceptance": "not_performed"})
exit_status = 0 if status == 0 and before == after and not findings else 2
print(json.dumps({"candidate_run": str(target), "observation_run": str(observation), "exit_status": exit_status}))
raise SystemExit(exit_status)
```
